val_ds_full = tf.keras.utils.image_dataset_from_directory(
    data_path,
    validation_split=0.2, # 20% for validation/test
    subset="validation",
    seed=SEED,
    image_size=IMG_SIZE,
    batch_size=BATCH_SIZE,

val_batches = tf.data.experimental.cardinality(val_ds_full)
test_ds = val_ds_full.take(val_batches // 2)

print(f"Validation batches: {tf.data.experimental.cardinality(val_ds)}")

# 2.4. Define Data Augmentation & Preprocessing
data_augmentation = tf.keras.Sequential([
    tf.keras.layers.RandomFlip("horizontal"),
    tf.keras.layers.RandomRotation(0.2),
])

AUTOTUNE = tf.data.AUTOTUNE
def prepare_data(ds, augment=False):
    ds = ds.map(lambda x, y: (tf.keras.applications.resnet50.preprocess_input(x), y),
                num_parallel_calls=AUTOTUNE)
    if augment:
        ds = ds.map(lambda x, y: (data_augmentation(x, training=True), y),
                    num_parallel_calls=AUTOTUNE)
    return ds.cache().prefetch(buffer_size=AUTOTUNE)

# 2.5. Apply Preprocessing
train_ds = prepare_data(train_ds, augment=True)
val_ds = prepare_data(val_ds)
test_ds = prepare_data(test_ds)

import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, GlobalAveragePooling2D, Dense, Dropout
# This function builds our models
def build_resnet50_model(num_classes, final_activation):
    base_model = tf.keras.applications.ResNet50(
        include_top=False,
        weights='imagenet',
        input_shape=(IMG_SIZE[0], IMG_SIZE[1], 3)
    )
    base_model.trainable = False
    inputs = Input(shape=(IMG_SIZE[0], IMG_SIZE[1], 3))
    x = base_model(inputs, training=False)
    x = GlobalAveragePooling2D()(x)
    x = Dense(512, activation='relu', name='features')(x) # Named 'features'
    x = Dropout(0.5)(x)
    outputs = Dense(num_classes, activation=final_activation)(x) 
    model = Model(inputs, outputs)
    return model
# 1. Build the SIGMOID model
model_sigmoid = build_resnet50_model(num_classes, 'sigmoid')
# 2. Compile the SIGMOID model
model_sigmoid.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
    loss='binary_crossentropy',
    metrics=['accuracy']
)
print("--- Model 1: ResNet50 + Sigmoid ---")
model_sigmoid.summary()
# 3. Define Checkpoint
checkpoint_path_sigmoid = "/content/drive/MyDrive/models/mango_disease/resnet50_sigmoid.weights.h5"
os.makedirs(os.path.dirname(checkpoint_path_sigmoid), exist_ok=True)
checkpoint_callback_sigmoid = tf.keras.callbacks.ModelCheckpoint(
    filepath=checkpoint_path_sigmoid,
    monitor='val_accuracy',
    save_best_only=True,
    save_weights_only=True,
    verbose=1
)
# 4. Train for 5 Epochs
print("\nStarting Sigmoid model training for 5 epochs...")
history_sigmoid = model_sigmoid.fit(
    train_ds,
    epochs=5,
    validation_data=val_ds,
    callbacks=[checkpoint_callback_sigmoid]
)
print("\nSigmoid model training complete.")

# Function to plot training history
def plot_history(history, title):
    acc = history.history['accuracy']
    val_acc = history.history['val_accuracy']
    loss = history.history['loss']
    val_loss = history.history['val_loss']
    plt.figure(figsize=(12, 5))
    plt.subplot(1, 2, 1)
    plt.plot(acc, label='Training Accuracy')
    plt.plot(val_acc, label='Validation Accuracy')
    plt.legend(loc='lower right')
    plt.title(f'{title} - Accuracy')
    plt.xlabel('Epoch')
    plt.subplot(1, 2, 2)
    plt.plot(loss, label='Training Loss')
    plt.plot(val_loss, label='Validation Loss')
    plt.legend(loc='upper right')
    plt.title(f'{title} - Loss')
    plt.xlabel('Epoch')
    
    plt.tight_layout()
    plt.show()
